from Snowflake_Dataframe import conn as snowflake_conn  # <-- Use Snowflake connection
import pandas as pd

class DataLoader:
    ...  # Keep all existing methods

    @staticmethod
    def read_snowflake(query: str) -> pd.DataFrame:
        """Read data from Snowflake using a SQL query."""
        try:
            df = pd.read_sql(query, snowflake_conn)
            if df.empty:
                raise ValueError("No data returned from Snowflake")
            return df
        except Exception as e:
            raise RuntimeError(f"Failed to read from Snowflake: {str(e)}")
